  <script>

    async function enviarPergunta(){

      const apiKey = document.getElementById("apiKey").value;
      const pergunta = document.getElementById("pergunta").value;
      const respostaDiv = document.getElementById("resposta");

      if(!apiKey){
        alert("Digite sua chave da API");
        return;
      }

      if(!pergunta){
        alert("Digite uma pergunta");
        return;
      }

      respostaDiv.innerHTML = "Carregando...";

      try{

        const response = await fetch("https://api.openai.com/v1/responses", {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
            "Authorization": `Bearer ${apiKey}`
          },

          body: JSON.stringify({

            model: "gpt-4.1-nano",

            input: [
              {
                role: "system",
                content: [
                  {
                    type: "input_text",
                    text: `
Responde apenas perguntas relacionadas a NR-1.

Perguntas sem relação a NR-1 devem ser respondidas apenas com:
"desculpe, apenas perguntas relacionadas a NR-1"

As perguntas devem ser respondidas de forma:
- concisa
- analítica
- formal
                    `
                  }
                ]
              },

              {
                role: "user",
                content: [
                  {
                    type: "input_text",
                    text: pergunta
                  }
                ]
              }
            ],

            tools: [
              {
                type: "file_search",
                vector_store_ids: [
                  "vs_69ff84c41e28819186587d6482f6f8b6"
                ]
              }
            ],

            temperature: 1,
            max_output_tokens: 2048,
            top_p: 1

          })

        });

        const data = await response.json();

        console.log(data);

        // PEGA TEXTO DA RESPOSTA
        const texto = data.output_text || "Sem resposta.";

        respostaDiv.innerHTML = texto;

      }catch(error){

        console.error(error);

        respostaDiv.innerHTML = 
          "Erro ao conectar com a OpenAI.";

      }

    }

  </script>
